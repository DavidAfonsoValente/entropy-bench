\begin{frame}
  \frametitle{Why this is not already how it is done}

  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{0.7em}
    \item The closest prior test (50 models of 1B): perplexity on a general held-out set orders
      fine-tuned pairs \warmw{worse than chance}; few-shot accuracy does best.
    \item Two details decide it, and the usual practice gets both wrong:
      \begin{enumerate}\setlength\itemsep{0.35em}
        \item \keyw{Which text}: it has to be the task's domain.
        \item \keyw{How it is cut}: whole documents, each from its own start, not fixed blocks
          from a concatenated stream.
      \end{enumerate}
  \end{itemize}
\end{frame}

% ==================================================================== 8
\begin{frame}
  \frametitle{No training, and no noise}

  \vspace{0.5em}
  \begin{columns}[t]
  \begin{column}{0.47\textwidth}
    \keyw{Adapting first changes little}
    \begin{itemize}\setlength\itemsep{0.4em}\small
      \item A short LoRA adaptation lowers every model's loss by 2.0--7.9\% on news.
      \item The document ranking barely moves: Spearman \goodw{0.993}, 38 of 39 pairs instead of 37.
      \item It does repair the block reading; that is how we found the problem.
    \end{itemize}
  \end{column}
  \begin{column}{0.47\textwidth}
    \keyw{Deterministic}
    \begin{itemize}\setlength\itemsep{0.4em}\small
      \item Re-running the reading reproduces every model's \bpb{} exactly.
      \item Same weights, new host: GSM8K moves \warmw{$-0.60$ to $+0.69$} points; one shipped
        decoding setting moves Gemma-4-12B's MMLU-Pro by \warmw{14.30}.
    \end{itemize}
  \end{column}
  \end{columns}
\end{frame}

% ==================================================================== 9
\begin{frame}
  \frametitle{Why loss is the right quantity}

  \vspace{0.7em}
  \begin{itemize}\setlength\itemsep{0.8em}
    \item A base model acquires its capabilities by minimising cross-entropy on text; fine-tuning
      minimises it again on the task.
    \item Cross-entropy is a proper scoring rule. Over the domain's text $p$:
      \[ H(p,q) \;=\; \underbrace{H(p)}_{\text{same for every model}} \;+\; \mathrm{KL}(p\,\|\,q) \]
    \item Per \keyw{byte}, so every tokenizer is on one scale; on text \keyw{dated after every
      release}, so no candidate has trained on it.
  \end{itemize}
\end{frame}

% ==================================================================== 10
\begin{frame}
  \frametitle{Using it, and adding your model}

  \vspace{0.5em}
  \begin{itemize}\setlength\itemsep{0.55em}
    \item \texttt{entropy-bench --baseline-only} reports \texttt{doc\_bpb}: one forward pass per
      candidate over a few MB of the domain's text.
    \item Text from the task's domain, collected after the candidates shipped.
  \end{itemize}

  \vspace{0.4em}
  \small
  \textbf{Public leaderboard, document tracks:} \keyw{arXiv maths} (CC0 corpus in the repo; anyone
  can run it and submit) and \keyw{news} (maintainers run requested models), 17 models each.
  Submissions are checked against the track's corpus hash, context, stride and scored bytes.
\end{frame}
