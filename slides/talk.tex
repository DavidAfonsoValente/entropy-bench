\begin{frame}
  \frametitle{Two reasons the scorecard misleads}

  \vspace{0.6em}
  \begin{columns}[t]
  \begin{column}{0.47\textwidth}
    \keyw{1. Benchmarks are public}
    \begin{itemize}\setlength\itemsep{0.4em}
      \item Written before the model existed, for other purposes.
      \item Vendors optimise for the ones they report.
      \item Items and subject matter can reach pre-training data.
      \item Stronger model, or saw more of the exam? \\ No reader can tell.
    \end{itemize}
  \end{column}
  \begin{column}{0.47\textwidth}
    \keyw{2. They score a model before it is adapted}
    \begin{itemize}\setlength\itemsep{0.4em}
      \item Teams adapt a base model to their data first.
      \item What matters is how good it \emph{becomes}.
      \item \textbf{Gemma-4-12B}, of fifteen models: 9th on GSM8K, 8th on MMLU-Pro,
        13th on raw loss \ldots
      \item \ldots\ and \goodw{the best fine-tuned system of all}.
    \end{itemize}
  \end{column}
  \end{columns}
\end{frame}

% ==================================================================== 4
\begin{frame}
  \frametitle{Why loss is the right quantity}
  \framesubtitle{There is no universal ground truth for ``better'' --- so pick the principled yardstick}

  \vspace{0.7em}
  \begin{itemize}\setlength\itemsep{0.8em}
    \item Every base model was trained to minimise cross-entropy on text. Held-out loss is
      \keyw{its own objective}, measured out of sample.
    \item Cross-entropy is a proper scoring rule. In expectation over the domain's text $p$:
      \[ H(p,q) \;=\; \underbrace{H(p)}_{\text{same for every model}} \;+\; \mathrm{KL}(p\,\|\,q) \]
      so a difference in loss \emph{is} a difference in distance from the text's distribution.
    \item The one obstacle: each model counts in its own tokens. Divide by \keyw{bytes}.
  \end{itemize}

  \punch{Bits per byte is not one metric among many. It is the training objective, put on a
  common scale.}
\end{frame}

  \begin{center}

    \node[box] (a) {\textbf{Bytes}\\[2pt]\scriptsize every tokenizer on one scale};
    \node[box, right=of a] (b) {\textbf{Fresh, decontaminated text}\\[2pt]\scriptsize no candidate has trained on it};
    \node[box, right=of b] (c) {\textbf{Equal adaptation}\\[2pt]\scriptsize the same procedure for every candidate};
  \end{tikzpicture}
  \end{center}

  \vspace{0.5em}
  \small
  \textbf{Decontamination, three layers:}
  \begin{itemize}\setlength\itemsep{0.25em}
    \item \keyw{Time} --- text dated after every model shipped (Reddit and Hacker News, most of arXiv).
    \item \keyw{Hygiene} --- exact dedup, quality filter, MinHash near-dup removal, 80/10/10 split with no overlap.
    \item \keyw{Membership screens} --- Min-K++, context sensitivity, perturbation; the union of
      anything any model flags is quarantined for all (news snapshot).
  \end{itemize}
\end{frame}

% ==================================================================== 6
\begin{frame}
  \frametitle{Equal adaptation: the procedure, and what we ran}

  \vspace{0.5em}
  \begin{columns}[t]
  \begin{column}{0.48\textwidth}
    \keyw{The method} \hfill{\scriptsize\color{mute}\texttt{--hparams sweep}, default}
    \begin{itemize}\setlength\itemsep{0.35em}\small
      \item Per-model Optuna search: TPE sampler over LR, batch, warm-up, weight decay,
        LoRA rank, $\alpha$, dropout.
      \item Successive halving prunes weak trials from 50 steps.
      \item Scored on the \emph{validation} split; test never touched.
      \item Same number of trials for every model --- no time cap.
      \item Winner trained until validation \bpb{} stops improving.
    \end{itemize}
  \end{column}
  \begin{column}{0.48\textwidth}
    \keyw{What these results use} \hfill{\scriptsize\color{mute}\texttt{--hparams manual}}
    \begin{itemize}\setlength\itemsep{0.35em}\small
      \item A full search of 17 models up to 35B was beyond our compute.
      \item One hand-picked recipe: LoRA $r{=}16$, $\alpha{=}32$, lr $10^{-4}$, batch 32,
        fixed 250 steps.
      \item One A100: \goodw{3.7 min} at 0.5B, 11.6 at 7.7B, 24.7 at 12B, 62.7 at 31B.
      \item Where our earlier search settled: rank 16 won for 8 of 11 models.
    \end{itemize}
  \end{column}
  \end{columns}

  \vspace{0.6em}
  \punch{\small A time-capped search is a trap: small models finish more trials and get better
  tuned. In our first run the 1B models completed 17--18 trials, the two largest only 4.}
\end{frame}

\begin{frame}
  \frametitle{Independent views, not ground truth}
  \framesubtitle{A task shows which model is better \emph{at that task}. So we ask: does the loss agree with uses it never saw?}

  \vspace{0.6em}
  \begin{columns}[t]
  \begin{column}{0.47\textwidth}
    \keyw{One real use: fine-tune them}
    \begin{itemize}\setlength\itemsep{0.35em}\small
      \item Do what a team does: fine-tune 15 models on a real news task (write the lede).
      \item ROUGE-L on 500 articles no measurement touched.
      \item If this were the only task, fine-tuning on it would be the evaluation --- but it needs labels, budget, and speaks for one task.
    \end{itemize}
  \end{column}
  \begin{column}{0.47\textwidth}
    \keyw{A quiz no model has seen}
    \begin{itemize}\setlength\itemsep{0.35em}\small
      \item Hide a name or number in a held-out article; the model writes it.
      \item Answered by the \emph{released} weights, not the adapted ones.
      \item \emph{Lenient}: answer starts in the first 24 characters (primary). \emph{Strict}: exact start.
    \end{itemize}
  \end{column}
  \end{columns}

  \vspace{0.8em}
  \punch{On news the two views agree at Spearman \keyw{0.921}; the quiz carries the comparison to the
  other three corpora. ``Right'' below means \emph{agrees with the view}.}
\end{frame}

  equal adaptation removes the familiarity.}

\begin{frame}
  \frametitle{Using it}

  \vspace{0.7em}
  \begin{enumerate}\setlength\itemsep{0.7em}
    \item Candidates more than $\sim$2$\times$ apart in size? The larger is usually better.
    \item Otherwise: gather recent text from your domain, dated after every candidate shipped, and
      screen it.
    \item Adapt every finalist by the same procedure --- the per-model search where compute
      allows, the hand-picked recipe where it does not --- and rank by adapted \bpb{}.
    \item Treat a gap under \keyw{2\%} as a tie; then licence, latency or cost decide.
    \item The zero-shot number is a free screen, not the selector.
  \end{enumerate}

  \punch{\small No labels. No answer key. Text you own, that nobody could have trained on.}
\end{frame}
